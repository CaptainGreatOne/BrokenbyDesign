**1. Create `prometheus/alerts/service-alerts.yml`:**

Create the `prometheus/alerts/` directory and the alert rules file:

```yaml
groups:
  - name: service_health
    interval: 15s
    rules:
      # 1. InstanceDown - Service completely unreachable by Prometheus
      - alert: InstanceDown
        expr: up == 0
        for: 1m
        labels:
          severity: critical
          service: "{{ $labels.job }}"
        annotations:
          summary: "Instance {{ $labels.instance }} is down"
          description: "{{ $labels.job }} instance {{ $labels.instance }} has been unreachable for more than 1 minute."

      # 2. ServiceUnhealthy - Instance reachable but health check fails
      - alert: ServiceUnhealthy
        expr: service_healthy == 0
        for: 1m
        labels:
          severity: critical
          service: "{{ $labels.job }}"
        annotations:
          summary: "Service {{ $labels.job }} is unhealthy"
          description: "{{ $labels.job }} instance {{ $labels.instance }} is running but reporting unhealthy status."

      # 3. HighErrorRate - Too many 5xx responses (>5% for 5 minutes)
      - alert: HighErrorRate
        expr: |
          (
            sum by (job) (rate(http_requests_total{status_code=~"5.."}[5m]))
            /
            sum by (job) (rate(http_requests_total[5m]))
          ) > 0.05
        for: 5m
        labels:
          severity: warning
          service: "{{ $labels.job }}"
        annotations:
          summary: "High error rate on {{ $labels.job }}"
          description: "{{ $labels.job }} has {{ $value | humanizePercentage }} 5xx errors (threshold: 5%)."

      # 4. HighLatency - p95 latency exceeds 1 second for 5 minutes
      - alert: HighLatency
        expr: |
          histogram_quantile(0.95,
            sum by (job, le) (rate(http_request_duration_seconds_bucket[5m]))
          ) > 1
        for: 5m
        labels:
          severity: warning
          service: "{{ $labels.job }}"
        annotations:
          summary: "High latency on {{ $labels.job }}"
          description: "{{ $labels.job }} p95 latency is {{ $value | humanizeDuration }} (threshold: 1s)."
```

IMPORTANT notes on PromQL expressions:
- HighErrorRate uses `status_code` label (matching web-gateway's `http_requests_total` metric). Order-api uses gRPC metrics (`grpc_requests_total` with `status` label) which won't match this HTTP rule — that's OK because the web-gateway is the HTTP frontend. The alert naturally applies only to services with HTTP metrics.
- HighLatency uses `http_request_duration_seconds_bucket` which only web-gateway exposes. This is correct — web-gateway is the HTTP entry point.
- ServiceUnhealthy uses the `service_healthy` gauge added in Plan 01.
- InstanceDown uses the auto-generated `up` metric.

**2. Create `alertmanager/alertmanager.yml`:**

Create the `alertmanager/` directory and config:

```yaml
global:
  resolve_timeout: 5m

route:
  receiver: 'all-receivers'
  group_by: ['alertname', 'service']
  group_wait: 10s
  group_interval: 5m
  repeat_interval: 4h

  routes:
    - match:
        severity: critical
      receiver: 'critical-receivers'
      group_wait: 5s
      group_interval: 2m
      repeat_interval: 1h

receivers:
  - name: 'all-receivers'
    webhook_configs:
      - url: 'http://webhook-receiver:5001/alerts'
        send_resolved: true
      - url: 'http://mock-slack-ui:8080/webhook'
        send_resolved: true

  - name: 'critical-receivers'
    webhook_configs:
      - url: 'http://webhook-receiver:5001/alerts'
        send_resolved: true
      - url: 'http://mock-slack-ui:8080/webhook'
        send_resolved: true

inhibit_rules: []
```

Add three new services to `docker-compose.yml` in the Observability Services section, after the existing `grafana` service:

**alertmanager service:**
```yaml
  alertmanager:
    image: prom/alertmanager:v0.28.1
    container_name: alertmanager
    restart: unless-stopped
    volumes:
      - ./alertmanager/alertmanager.yml:/etc/alertmanager/alertmanager.yml:ro
      - alertmanager-data:/alertmanager
    command:
      - '--config.file=/etc/alertmanager/alertmanager.yml'
      - '--storage.path=/alertmanager'
      - '--web.external-url=http://localhost:9093'
    ports:
      - "9093:9093"
    depends_on:
      prometheus:
        condition: service_healthy
    healthcheck:
      test: ["CMD-SHELL", "wget --spider -q http://localhost:9093/-/healthy || exit 1"]
      interval: 5s
      timeout: 3s
      retries: 3
      start_period: 10s
    deploy:
      resources:
        limits:
          memory: 256M
          cpus: '0.5'
    logging:
      driver: json-file
      options:
        max-size: "10m"
        max-file: "3"
```

Note on Alertmanager version: Use v0.28.1 (confirmed stable, available on Docker Hub). The research mentioned v0.31.0 but verify availability. If v0.28.1 is not available, use the latest stable tag. The key is to use a stable, recent version.
